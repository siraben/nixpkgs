cmake_minimum_required(VERSION 3.13)
project(ocio_minizip_finder_probe LANGUAGES C CXX)
set(OCIO_MODULE_DIR "" CACHE PATH "Patched OCIO share/cmake/modules directory")
option(FORCE_PKGCONFIG "Bypass only the finder's nested config-package search" OFF)
if(NOT EXISTS "${OCIO_MODULE_DIR}/Findminizip-ng.cmake")
    message(FATAL_ERROR "Set OCIO_MODULE_DIR to the proposed-patched OCIO finder directory")
endif()
find_package(ZLIB REQUIRED)
set(OCIO_INSTALL_EXT_PACKAGES NONE)
set(minizip-ng_FIND_VERSION 4.2.2)
if(FORCE_PKGCONFIG)
    # Include the module directly so disabling its optional nested find_package
    # does not disable an outer REQUIRED find_package call as well.
    set(CMAKE_DISABLE_FIND_PACKAGE_minizip-ng TRUE)
endif()
include("${OCIO_MODULE_DIR}/Findminizip-ng.cmake")
if(NOT TARGET MINIZIP::minizip-ng)
    message(FATAL_ERROR "OCIO finder did not create an installed minizip target")
endif()
set(probe_minizip_target MINIZIP::minizip-ng)
get_target_property(public_definitions ${probe_minizip_target} INTERFACE_COMPILE_DEFINITIONS)
get_target_property(public_options ${probe_minizip_target} INTERFACE_COMPILE_OPTIONS)
message(STATUS "OCIO finder definitions: ${public_definitions}; options: ${public_options}")
include(../common/probes.cmake)
if(MINIZIP_EXPECT_FILE_OFFSET_64)
    if(FORCE_PKGCONFIG)
        if(NOT PC_minizip-ng_FOUND OR NOT "-D_FILE_OFFSET_BITS=64" IN_LIST public_options)
            message(FATAL_ERROR "OCIO pkg-config fallback lost the public compilation flag")
        endif()
    elseif(NOT "_FILE_OFFSET_BITS=64" IN_LIST public_definitions)
        message(FATAL_ERROR "OCIO config-package path lost the public definition")
    endif()
endif()
