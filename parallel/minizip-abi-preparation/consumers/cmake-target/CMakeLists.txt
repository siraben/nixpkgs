cmake_minimum_required(VERSION 3.13)
project(minizip_installed_target_probe LANGUAGES C CXX)
option(MINIZIP_COMPAT "Consume the installed compatibility package" OFF)
if(MINIZIP_COMPAT)
    find_package(minizip 4.2.2 CONFIG REQUIRED)
    set(probe_minizip_target MINIZIP::minizip)
else()
    find_package(minizip-ng 4.2.2 CONFIG REQUIRED)
    set(probe_minizip_target MINIZIP::minizip-ng)
endif()
get_target_property(public_definitions ${probe_minizip_target} INTERFACE_COMPILE_DEFINITIONS)
message(STATUS "${probe_minizip_target} public definitions: ${public_definitions}")
include(../common/probes.cmake)
if(MINIZIP_EXPECT_FILE_OFFSET_64 AND NOT "_FILE_OFFSET_BITS=64" IN_LIST public_definitions)
    message(FATAL_ERROR "Installed CMake target lacks the public file-offset ABI contract")
endif()
