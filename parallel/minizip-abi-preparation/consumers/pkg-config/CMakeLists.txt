cmake_minimum_required(VERSION 3.13)
project(minizip_installed_pkgconfig_probe LANGUAGES C CXX)
option(MINIZIP_COMPAT "Consume the installed compatibility package" OFF)
find_package(PkgConfig REQUIRED)
if(MINIZIP_COMPAT)
    pkg_check_modules(MZ REQUIRED IMPORTED_TARGET "minizip>=4.2.2")
else()
    pkg_check_modules(MZ REQUIRED IMPORTED_TARGET "minizip-ng>=4.2.2")
endif()
set(probe_minizip_target PkgConfig::MZ)
message(STATUS "pkg-config compilation flags: ${MZ_CFLAGS}")
include(../common/probes.cmake)
if(MINIZIP_EXPECT_FILE_OFFSET_64 AND NOT "-D_FILE_OFFSET_BITS=64" IN_LIST MZ_CFLAGS_OTHER)
    message(FATAL_ERROR "Installed pkg-config Cflags lack the public file-offset ABI contract")
endif()
